  <script>
    const form = document.getElementById('uploadForm');
    const submitBtn = document.getElementById('submitBtn');
    const loader = document.getElementById('loader');
    const placeholder = document.getElementById('placeholder');
    const viewer = document.getElementById('viewer');
    const errorText = document.getElementById('errorText');

    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      
      const fileInput = document.getElementById('pdfFile');
      if (!fileInput.files[0]) return;

      const formData = new FormData();
      formData.append('file', fileInput.files[0]);
      formData.append('kind', document.getElementById('kind').value);
      formData.append('teeth', document.getElementById('teeth').value);
      formData.append('gonfiore', document.getElementById('gonfiore').value);

      // UI State: Loading
      loader.style.display = 'block';
      placeholder.style.display = 'none';
      errorText.style.display = 'none';
      submitBtn.disabled = true;

      try {
        const response = await fetch('/generate-3d', {
          method: 'POST',
          body: formData
        });

        if (!response.ok) {
          const errorData = await response.json().catch(() => ({ detail: 'Errore generico durante la conversione.' }));
          throw new Error(errorData.detail || 'Errore nella generazione del modello 3D.');
        }

        // Converte la risposta binaria GLB in un URL Blob per l'elemento <model-viewer>
        const blob = await response.blob();
        const glbUrl = URL.createObjectURL(blob);
        
        viewer.src = glbUrl;
      } catch (err) {
        errorText.textContent = err.message;
        errorText.style.display = 'block';
      } finally {
        loader.style.display = 'none';
        submitBtn.disabled = false;
      }
    });
  </script>
